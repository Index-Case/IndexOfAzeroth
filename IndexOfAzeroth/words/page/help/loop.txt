options = Options
options.li1 = <b>AOE</b> picks the densest packs at the character's level or below. AOE is meant for a Frost Mage from level 22 or a Protection Paladin from level 40.
options.li2 = <b>Include AH prices</b> counts loot at what the loot sells for in the auction house, less the house's 5% cut.
options.li3 = <b>Daredevil</b> lets a gathering loop go through every mob.
options.li4 = <b>All levels</b> lets a loop take mobs of any level, with no check for danger.
